- name: New parition 10 GB partition
  hosts: 172.31.13.78
  become: true
  tasks:
          #- name: create partition
          #parted:
          #device: /dev/sdc
          #number: 1
          #flags: lvm
          #state: present
          #part_end: 9GB

    - name: task for creating volume group
      lvg:
          vg: sample-vg
          pvs: /dev/sdd
          pesize: 4

    - name: task for creating logical volume
      lvol:
          vg: sample-vg
          lv: sample-lv
          size: 100%FREE
          shrink: true
          force: yes

    - name: Create directory data if does not exist
      file:
        path: /data
        state: directory
        mode: '0755'

    - name: format the xfs filesystem
      filesystem:
        fstype: xfs
        dev: /dev/sample-vg/sample-lv

    - name: mount the lv on /data
      mount:
        path: /data
        src: /dev/sample-vg/sample-lv
        fstype: xfs
        state: mounted
