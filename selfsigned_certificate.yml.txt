---
- hosts: 172.31.13.75
  become: yes
  tasks:
    - name: Install openssl package
      yum:
        pkg: openssl
        state: latest

    - name: Create ssl directory
      file:
        path: /etc/ssl/
        mode: '0775'
        recurse: yes
        state: directory

    - name: Generate Private key
      openssl_privatekey:
        path: /etc/ssl/blakshmi.key
        type: RSA
        size: 2048

    - openssl_csr:
        path: /etc/ssl/blakshmi.csr
        privatekey_path: /etc/ssl/blakshmi.key
        country_name: IN
        state_or_province_name: Karnataka
        locality_name: Bangalore
        organization_name: Department
        organizational_unit_name: gfg
        common_name: data.karnataka.gov.in

    - name: Generate a self signed certificate
      openssl_certificate:
        csr_path: /etc/ssl/blakshmi.csr
        path: /etc/ssl/blakshmi.crt
        privatekey_path: /etc/ssl/blakshmi.key
        provider: selfsigned
