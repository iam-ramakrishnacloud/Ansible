---
  - hosts: 172.31.13.78
    tasks:
    - name: create a partition
      parted:
        device: /dev/sdc
        number: 1
        flags: [lvm]
        state: present
        part_end: 20GB

    - name: create vg
      lvg:
        vg: sample_vg
        pvs: /dev/sdc1

    - name: create a lv
      lvol:
        vg: sample_vg
        lv: sample1_lv
        size: 100%FREE
        force: yes

    - name: create directory
      file:
        path: /data
        state: directory
    - name: filesystem xfs
      filesystem:
        fstype: xfs
        dev: /dev/sample_vg/sample1_lv

    - name: mount
      mount:
         path: /data
         src: /dev/sample_vg/sample1_lv
         fstype: xfs
         state: mounted
